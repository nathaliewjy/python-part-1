#  1. for loop integer : range(...), range(start, stop), range(start, stop, step)
for i in range(5):
    print(f"Halo ke-{i+1}")

for i in range(1, 6):
    print(i, end=" ")
print()
for i in range(1, 10, 2):
    print(i, end=" ")
print()

# 2. for loop string : with & without end=" "
for i in "Halo":
    print(i, end="")
print()

# 3. for loop else statement : with & without if break
for i in range(5):
    print(i)
else:
    print("For loop selesai")

for i in range(5):
    if i == 3:
        continue
    print(i)
else:
    print("For loop tidak selesai")
